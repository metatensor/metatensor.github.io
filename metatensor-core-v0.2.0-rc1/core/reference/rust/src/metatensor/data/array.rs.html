13/// The Array trait is used by metatensor to manage different kind of data array
14/// with a single API. Metatensor only knows about `Box<dyn Array>`, and
15/// manipulate the data through the functions on this trait.
16///
17/// This corresponds to the `mts_array_t` struct in metatensor-core.
18pub trait Array: std::any::Any + Send + Sync {
19    /// Get the array as a `Any` reference
20    fn as_any(&self) -> &dyn std::any::Any;
21
22    /// Get the array as a mutable `Any` reference
23    fn as_any_mut(&mut self) -> &mut dyn std::any::Any;
24
25    /// Create a new array with the same options as the current one (data type,
26    /// data location, etc.) and the requested `shape`.
27    ///
28    /// The new array should be filled with the scalar value from `fill_value`,
29    /// which must be an `MtsArray` with shape `(1,)` and the same dtype as
30    /// this array.
31    fn create(&self, shape: &[usize], fill_value: MtsArray) -> Box<dyn Array>;
32
33    /// Make a copy of this `array`
34    ///
35    /// The new array is expected to have the same data origin and parameters
36    /// (data type, data location, etc.)
37    fn copy(&self) -> Box<dyn Array>;
38
39    /// Get the shape of the array. This can be empty if the array has no shape
40    /// (e.g. a scalar).
41    fn shape(&self) -> Vec<usize>;
42
43    /// Change the shape of the array to the given `shape`
44    fn reshape(&mut self, shape: &[usize]);
45
46    /// Swap the axes `axis_1` and `axis_2` in this array
47    fn swap_axes(&mut self, axis_1: usize, axis_2: usize);
48
49    /// Set entries in `self` taking data from the `input` array.
50    ///
51    /// The `output` array is guaranteed to be created by calling
52    /// `mts_array_t::create` with one of the arrays in the same block or tensor
53    /// map as the `input`.
54    ///
55    /// The `movements` indicate where the data should be moved from `input` to
56    /// `output`.
57    ///
58    /// This function should copy data from `input[movements[i].sample_in, ...,
59    /// movements[i].properties_start_in + x]` to
60    /// `array[movements[i].sample_out, ..., movements[i].properties_start_out +
61    /// x]` for `i` up to `movements_count` and `x` up to
62    /// `movements[i].properties_length`. All indexes are 0-based.
63    fn move_data(
64        &mut self,
65        input: &dyn Array,
66        movements: &[mts_data_movement_t],
67    );
68
69    /// Get the device where this array's data resides.
70    ///
71    /// For CPU arrays this should return `DLDevice::cpu()`.
72    fn device(&self) -> DLDevice;
73
74    /// Get the data type of this array.
75    ///
76    /// This populates the `dtype` vtable slot for fast dtype queries.
77    /// Implementations should return the appropriate `DLDataType` for their
78    /// element type (e.g. float64 = `DLDataType { code: kDLFloat, bits: 64, lanes: 1 }`).
79    fn dtype(&self) -> DLDataType;
80
81    /// Convert the array to a `DLPack` tensor.
82    /// The returned pointer is owned by the caller (and cleaned up via its deleter).
83    fn as_dlpack(
84        &self,
85        device: DLDevice,
86        stream: Option<i64>,
87        max_version: DLPackVersion
88    ) -> Result<DLPackTensor, Error>;
89}

110impl From<Box<dyn Array>> for MtsArray {
111    fn from(value: Box<dyn Array>) -> Self {
112        let shape = value.shape();
113        let array = RustArray {
114            impl_: value,
115            shape,
116        };
117
118        let raw = mts_array_t {
119            ptr: Box::into_raw(Box::new(array)).cast(),
120            origin: Some(rust_array_origin),
121            device: Some(rust_array_device),
122            dtype: Some(rust_array_dtype),
123            as_dlpack: Some(rust_array_as_dlpack),
124            shape: Some(rust_array_shape),
125            reshape: Some(rust_array_reshape),
126            swap_axes: Some(rust_array_swap_axes),
127            create: Some(rust_array_create),
128            copy: Some(rust_array_copy),
129            destroy: Some(rust_array_destroy),
130            move_data: Some(rust_array_move_data),
131        };
132
133        return MtsArray::from_raw(raw);
134    }
135}

137impl<T> From<T> for MtsArray where T: Array + 'static {
138    fn from(value: T) -> Self {
139        let boxed = Box::new(value) as Box<dyn Array>;
140        return MtsArray::from(boxed);

144macro_rules! check_pointers {
145    ($pointer: ident) => {
146        if $pointer.is_null() {
147            panic!(
148                "got invalid NULL pointer for {} at {}:{}",
149                stringify!($pointer), file!(), line!()
150            );
151        }
152    };
153    ($($pointer: ident),* $(,)?) => {
154        $(check_pointers!($pointer);)*
155    }
156}

222/// Implementation of `mts_array_t.reshape` using `RustArray`
223#[allow(clippy::cast_possible_truncation)]
224unsafe extern "C" fn rust_array_reshape(
225    array: *mut c_void,
226    shape: *const usize,
227    shape_count: usize,
228) -> mts_status_t {
229    crate::errors::catch_unwind(|| {
230        check_pointers!(array);
231        let array = array.cast::<RustArray>();
232
233        let shape = if shape_count == 0 {
234            &[]
235        } else {
236            check_pointers!(shape);
237            std::slice::from_raw_parts(shape, shape_count)
238        };
239
240        (*array).impl_.reshape(shape);
241        (*array).shape = shape.to_vec();
242
243        Ok(())
244    })
245}

310/// Implementation of `mts_array_t.destroy` for `RustArray`

312    array: *mut c_void,
313) {
314    if !array.is_null() {

316        let boxed = Box::from_raw(array);
317        std::mem::drop(boxed);
318    }

321/// Implementation of `mts_array_t.move_sample` using `RustArray`
322#[allow(clippy::cast_possible_truncation)]
323unsafe extern "C" fn rust_array_move_data(
324    output: *mut c_void,
325    input: *const c_void,
326    movements: *const mts_data_movement_t,
327    movements_count: usize,
328) -> mts_status_t {
329    crate::errors::catch_unwind(|| {
330        check_pointers!(output, input);
331        let output = output.cast::<RustArray>();
332        let input = input.cast::<RustArray>();
333
334        let movements = if movements_count == 0 {
335            &[]
336        } else {
337            check_pointers!(movements);
338            std::slice::from_raw_parts(movements, movements_count)
339        };
340
341        (*output).impl_.move_data(&*(*input).impl_, movements);
342
343        Ok(())
344    })
345}

347/// Implementation of `mts_array_t.as_dlpack` using `RustArray`
348unsafe extern "C" fn rust_array_as_dlpack(
349    array: *mut c_void,
350    out: *mut *mut DLManagedTensorVersioned,
351    device: DLDevice,
352    stream: *const i64,
353    max_version: DLPackVersion,
354) -> mts_status_t {
355    crate::errors::catch_unwind(|| {
356        check_pointers!(array, out);
357        let array = array.cast::<RustArray>();
358        let stream_opt = stream.as_ref().copied();
359        let tensor = (*array).impl_.as_dlpack(device, stream_opt, max_version)?;
360
361        *out = tensor.into_raw().as_ptr();
362
363        Ok(())
364    })
