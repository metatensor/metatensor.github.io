11impl<T> From<ndarray::ArrayD<T>> for MtsArray where T: 'static + Clone + Send + Default + Sync + GetDLPackDataType + DLPackPointerCast {
12    fn from(value: ndarray::ArrayD<T>) -> Self {
13        let array = Arc::new(RwLock::new(value));
14        let boxed: Box<dyn Array> = Box::new(array);
15        return MtsArray::from(boxed);
16    }

19impl<T> Array for Arc<RwLock<ndarray::ArrayD<T>>>
20where
21    T: 'static + Send + Sync + Clone + Default + GetDLPackDataType + DLPackPointerCast,
22{
23    fn as_any(&self) -> &dyn std::any::Any {
24        self
25    }
26
27    fn as_any_mut(&mut self) -> &mut dyn std::any::Any {
28        self
29    }

31    fn create(&self, shape: &[usize], fill_value: MtsArray) -> Box<dyn Array> {
32        let cpu_device = DLDevice::cpu();
33        let max_version = DLPackVersion::current();
34        let fill_value_dlpack = fill_value.as_dlpack(cpu_device, None, max_version).expect("failed to extract fill_value as DLPack");
35
36        // Validate fill_value shape from the DLPack tensor directly
37        assert_eq!(fill_value_dlpack.shape(), &[], "fill_value must be a single scalar");
38        assert_eq!(fill_value_dlpack.device(), cpu_device, "fill_value must be on CPU");
39
40        let fill_value_ptr = fill_value_dlpack.data_ptr::<T>().expect("dtype mismatch between array and fill_value");
41        let fill_value_scalar = unsafe { std::ptr::read(fill_value_ptr) };
42
43        let array = ndarray::Array::from_elem(shape, fill_value_scalar);
44        return Box::new(Arc::new(RwLock::new(array)));
45    }
46
47    fn copy(&self) -> Box<dyn Array> {
48        return Box::new(self.clone());
49    }
50
51    fn shape(&self) -> Vec<usize> {
52        match self.try_read() {
53            Ok(lock) => lock.shape().to_vec(),
54            Err(TryLockError::Poisoned(_)) => panic!("array lock is poisoned"),
55            Err(TryLockError::WouldBlock) => panic!("array is already locked"),
56        }
57    }
58
59    fn reshape(&mut self, shape: &[usize]) {
60        let mut lock = match self.try_write() {
61            Ok(lock) => lock,
62            Err(TryLockError::Poisoned(_)) => panic!("array lock is poisoned"),
63            Err(TryLockError::WouldBlock) => panic!("array is already locked"),
64        };
65        let array = std::mem::take(&mut *lock);
66        let array = array.into_shape_clone(shape).expect("invalid shape");
67        let _ = std::mem::replace(&mut *lock, array);
68    }
69
70    fn swap_axes(&mut self, axis_1: usize, axis_2: usize) {
71        let mut lock = match self.try_write() {
72            Ok(lock) => lock,
73            Err(TryLockError::Poisoned(_)) => panic!("array lock is poisoned"),
74            Err(TryLockError::WouldBlock) => panic!("array is already locked"),
75        };
76        lock.swap_axes(axis_1, axis_2);
77    }

79    fn move_data(
80        &mut self,
81        input: &dyn Array,
82        movements: &[mts_data_movement_t],
83    ) {
84        use ndarray::{Axis, Slice};
85
86        let input = input.as_any().downcast_ref::<Self>().expect("input must be a ndarray of the same type");
87        let input = match input.try_read() {
88            Ok(lock) => lock,
89            Err(TryLockError::Poisoned(_)) => panic!("input array lock is poisoned"),
90            Err(TryLockError::WouldBlock) => panic!("input array is already locked"),
91        };
92
93        let mut output = match self.try_write() {
94            Ok(lock) => lock,
95            Err(TryLockError::Poisoned(_)) => panic!("output array lock is poisoned"),
96            Err(TryLockError::WouldBlock) => panic!("output array is already locked"),
97        };
98
99        if movements.is_empty() {
100            return;
101        }
102
103        // Check if we can use the optimized path (all moves have same property structure)
104        let first_prop_start_in = movements[0].properties_start_in;
105        let first_prop_start_out = movements[0].properties_start_out;
106        let first_prop_len = movements[0].properties_length;
107
108        let mut constant_properties = true;
109        let mut contiguous_input_samples = true;
110        let mut contiguous_output_samples = true;
111
112        for w in movements.windows(2) {
113            if w[0].properties_start_in != first_prop_start_in ||
114               w[0].properties_start_out != first_prop_start_out ||
115               w[0].properties_length != first_prop_len {
116                constant_properties = false;
117                break;
118            }
119
120            if w[1].sample_in != w[0].sample_in + 1 {
121                contiguous_input_samples = false;
122            }
123
124            if w[1].sample_out != w[0].sample_out + 1 {
125                contiguous_output_samples = false;
126            }
127        }
128
129        if constant_properties {
130            let last = movements.last().unwrap();
131            if last.properties_start_in != first_prop_start_in ||
132               last.properties_start_out != first_prop_start_out ||
133               last.properties_length != first_prop_len {
134                constant_properties = false;
135            }
136        }
137
138        let property_axis = output.shape().len() - 1;
139
140        if constant_properties {
141            let input_slice_info = Slice::from(first_prop_start_in..(first_prop_start_in + first_prop_len));
142            let output_slice_info = Slice::from(first_prop_start_out..(first_prop_start_out + first_prop_len));
143
144            if contiguous_input_samples && contiguous_output_samples {
145                let sample_start_in = movements[0].sample_in;
146                let sample_start_out = movements[0].sample_out;
147                let sample_count = movements.len();
148
149                let input_samples = input.slice_axis(
150                    Axis(0),
151                    Slice::from(sample_start_in..(sample_start_in + sample_count))
152                );
153                let mut output_samples = output.slice_axis_mut(
154                    Axis(0),
155                    Slice::from(sample_start_out..(sample_start_out + sample_count))
156                );
157
158                let value = input_samples.slice_axis(Axis(property_axis), input_slice_info);
159                let mut output_location = output_samples.slice_axis_mut(Axis(property_axis), output_slice_info);
160
161                output_location.assign(&value);
162            } else {
163                for move_item in movements {
164                    let input_sample = input.index_axis(Axis(0), move_item.sample_in);
165                    let mut output_sample = output.index_axis_mut(Axis(0), move_item.sample_out);
166
167                    let value = input_sample.slice_axis(
168                        // property_axis - 1 because we are slicing the sample
169                        // axis out, so the property axis is now one less
170                        Axis(property_axis - 1),
171                        input_slice_info
172                    );
173                    let mut output_location = output_sample.slice_axis_mut(
174                        Axis(property_axis - 1),
175                        output_slice_info
176                    );
177                    output_location.assign(&value);
178                }
179            }
180        } else {
181            // fallback to the general case
182            for move_item in movements {
183                let input_sample = input.index_axis(Axis(0), move_item.sample_in);
184                let mut output_sample = output.index_axis_mut(Axis(0), move_item.sample_out);
185
186                let value = input_sample.slice_axis(
187                    // see above for property_axis - 1 explanation
188                    Axis(property_axis - 1),
189                    Slice::from(move_item.properties_start_in..(move_item.properties_start_in + move_item.properties_length))
190                );
191                let mut output_location = output_sample.slice_axis_mut(
192                    Axis(property_axis - 1),
193                    Slice::from(move_item.properties_start_out..(move_item.properties_start_out + move_item.properties_length))
194                );
195                output_location.assign(&value);
196            }
197        }
198    }

208    fn as_dlpack(
209        &self,
210        device: DLDevice,
211        stream: Option<i64>,
212        max_version: DLPackVersion,
213    ) -> Result<DLPackTensor, Error> {
214        if stream.is_some() {
215            // we only support CPU for now
216            return Err(Error {
217                code: Some(crate::c_api::MTS_INVALID_PARAMETER_ERROR),
218                message: "CPU arrays can not be used with a stream".into(),
219            });
220        }
221        let vendored_version = DLPackVersion::current();
222        let major_mismatch = max_version.major != vendored_version.major;
223        let minor_too_high = max_version.minor < vendored_version.minor;
224        if major_mismatch || minor_too_high {
225            return Err(Error {
226                code: Some(crate::c_api::MTS_INVALID_PARAMETER_ERROR),
227                message: format!(
228                    "Metatensor supports DLPack version {}.{}. Caller requested incompatible version {}.{}",
229                    vendored_version.major, vendored_version.minor, max_version.major, max_version.minor
230                ),
231            });
232        }
233
234        let ndarray_device = DLDevice::cpu();
235
236        if device.device_type != ndarray_device.device_type || device.device_id != ndarray_device.device_id {
237            return Err(Error {
238                code: Some(crate::c_api::MTS_INVALID_PARAMETER_ERROR),
239                message: format!(
240                    "Requested DLPack device ({}) does not match array device ({})",
241                    device, ndarray_device
242                ),
243            });
244        }
245
246        let tensor: DLPackTensor = Arc::clone(self).try_into().map_err(|e| Error {
247            code: Some(crate::c_api::MTS_INVALID_PARAMETER_ERROR),
248            message: format!("failed to convert ndarray to DLPack: {:?}", e),
249        })?;
250
251        Ok(tensor)
252    }

289    fn ndarray_all_dtypes() {
290        fn test_for_dtype<T>(code: DLDataTypeCode, bits: u8) where T: Send + Sync + Clone + Default + GetDLPackDataType + DLPackPointerCast + 'static {
291            let data = ndarray::Array::<T, _>::from_elem(vec![2, 2], T::default());
292            let mts_array = MtsArray::from(data);
293
294            assert_eq!(mts_array.shape().unwrap(), [2, 2]);
295
296            // Should be able to export as DLPack
297            let dl_managed = mts_array.as_dlpack(DLDevice::cpu(), None, DLPackVersion::current()).unwrap();
298            assert_eq!(dl_managed.dtype().code, code);
299            assert_eq!(dl_managed.dtype().bits, bits);
300            assert_eq!(dl_managed.dtype().lanes, 1);
301
302
303            // And `create` should make an array of the same type (i32)
304            let fill_value = MtsArray::from(ndarray::Array::from_elem(vec![], T::default()));
305
306            let created = mts_array.create(&[1, 1], fill_value.as_ref()).unwrap();
307            let dl_managed = created.as_dlpack(DLDevice::cpu(), None, DLPackVersion::current()).unwrap();
308
309            assert_eq!(dl_managed.dtype().code, code);
310            assert_eq!(dl_managed.dtype().bits, bits);
311            assert_eq!(dl_managed.dtype().lanes, 1);
312        }
313
314        test_for_dtype::<bool>(DLDataTypeCode::kDLBool, 8);
315        test_for_dtype::<f64>(DLDataTypeCode::kDLFloat, 64);
316        test_for_dtype::<f32>(DLDataTypeCode::kDLFloat, 32);
317        test_for_dtype::<i8>(DLDataTypeCode::kDLInt, 8);
318        test_for_dtype::<i16>(DLDataTypeCode::kDLInt, 16);
